# =============================================================================
# Phase 1a validation config — deepstream-app single-stream visual gate.
#
# Runs the converted goec_N_v1 engine on a test clip and writes an ANNOTATED
# mp4 (boxes + labels) so you can eyeball that detections + class names are
# correct BEFORE building the full service pipeline (Phase 1b).
#
# Run from the DeepStream-Yolo work dir so the nvinfer relative paths resolve:
#   cd deepstream/DeepStream-Yolo
#   deepstream-app -c ../deepstream_validate.txt
#
# EDIT before running:
#   [source0] uri  -> absolute file:// path to a test clip
#   [sink0]  output-file -> where to write the annotated mp4
# Headless server: sink type=3 (encode to file). Do NOT use EGL/display sinks.
# =============================================================================

[application]
enable-perf-measurement=1
perf-measurement-interval-sec=5

[tiled-display]
enable=1
rows=1
columns=1
width=1280
height=720
gpu-id=0

[source0]
enable=1
# 3 = URI source
type=3
# EDIT: absolute path to a test clip (gun_front.mp4 exercises the 'gun' class)
uri=file:///REPLACE/WITH/ABS/PATH/services/camera-detection/test_videos/gun_front.mp4
num-sources=1
gpu-id=0
cudadec-memtype=0

[sink0]
enable=1
# 3 = encode to file (headless-safe). container=1 mp4, codec=1 H264.
# enc-type=0 = hardware NVENC (T4). Do NOT use enc-type=1 (software) — the
# DeepStream container lacks libavcodec/x264, so the sw encoder fails to create.
type=3
container=1
codec=1
enc-type=0
sync=0
bitrate=4000000
output-file=/tmp/goec_validate_out.mp4

[osd]
enable=1
gpu-id=0
border-width=3
text-size=15
text-color=1;1;1;1
text-bg-color=0.3;0.3;0.3;1
font=Serif
show-clock=0

[streammux]
gpu-id=0
batch-size=1
batched-push-timeout=40000
width=1920
height=1080
live-source=0
nvbuf-memory-type=0

[primary-gie]
enable=1
gpu-id=0
# nvinfer resolves a config's relative paths (onnx-file, custom-lib-path, ...)
# relative to the CONFIG FILE's own directory — NOT the CWD. So point at the copy
# staged inside DeepStream-Yolo/ (by convert_model.sh), where the ONNX, labels and
# parser .so all live, so those relative paths resolve. This path is itself
# resolved relative to THIS file's dir (deepstream/).
config-file=DeepStream-Yolo/config_infer_goec.txt
batch-size=1
gie-unique-id=1
nvbuf-memory-type=0

[tests]
file-loop=0
